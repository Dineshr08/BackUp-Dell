What we are doing here is based on the path we guve in browser we are manually redirecting the pages, all the pages we keep inside the pages folder then using fs operation we are doing the manual redirection. So firest we are check the path if /about then on path appending the full about.html path then using fs reading  the file content then then writing the response, in fs callback we have added data that will have content of .html file . if we want to the permenant redirection we need to handle using  res.setHeader("Location", "/"); if any user hit /home then thet need to redirect to /, so before that we need to mention res.statusCode = 301; as its redirecting and lso need to mention res.end();because after redirection we need to end the response then for / new req will gen and that will show the response in browser.on else part also if some dirrerent path given for that also we need to metion res.statusCode = 404. All this manual redirect can easily done using express js with route concept.
console.log(__filename);
const http = require("http");
const server = http.createServer((req, res) => {
  let path = './pages';
  console.log(req.url);
  if (req.url === "/") {
    path += "index.html";
  } else if (req.url === "/home") {
    res.statusCode = 301;
    res.setHeader("Location", "/");
    res.end();
  } else if (req.url === "/about") {
    path += "about.html";
  } else if (req.url == "/contact") {
    path += "contact.html";
  } else {
    path += "notFound.html";
    res.statusCode = 404;
  }

  const fs = require("fs");
  if (fs.existsSync(path)) {
    fs.readFile(path, (error, data) => {
      if (error) {
        console.log(error.message);
      } else {
        res.write(data);
        res.end();
      }
    });
  }
});

server.listen(3000, "localhost", () => {
  console.log("Server is listening...");
});
