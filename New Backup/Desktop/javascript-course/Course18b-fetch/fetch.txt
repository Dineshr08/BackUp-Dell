this is final code export function loadProductsFetch() {
  export function loadProductsFetch() {
  const promise = fetch('https://supersimplebackend.dev/products').then((response) => {
    return response.json(); // it become promise
  }).then((productData) => {

    products = productData.map((productDetails) => {
 
    if (productDetails.type === 'clothing') {
      return new Clothings(productDetails);
    }
    return new Products(productDetails);
  
})
  });
  return promise;
}
Here we store the res to promise variable and returning that so again we can use this function at whatever place we want and make .then and perform next step so the complete fn now become promise
